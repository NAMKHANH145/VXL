#include <stdio.h>
#include <string.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "esp_log.h"
#include "driver/i2c.h"
#include "driver/gpio.h"

#include "config.h"
#include "lcd.h"
#include "pwm.h"
#include "button.h"
#include "sht35.h"
#include "ds18b20.h"
#include "sd.h"
#include "fsm.h"
#include <inttypes.h>

#define TAG "VXL_MAIN"

// ================= GLOBAL VARS =================
SensorData_t g_sensors = {0};
SystemSetpoints_t g_setpoints = { .target_hum = DEFAULT_TARGET_HUM, .mode = 0 };
SystemState_t g_current_state = STATE_IDLE;

const char *STATE_NAMES[] = { "IDLE", "HUMID", "DEHUM", "DEFRT", "TEST", "EMERG" };

static bool is_setup_mode = false;

// ================= CONFIGURATION =================
#define SHT_SAMPLE_INTERVAL_MS  250   // Đọc SHT35 mỗi 250ms (4Hz)
#define MA_WINDOW_SIZE          20    // 20 mẫu * 250ms = 5 giây trung bình
#define LOG_INTERVAL_MS         5000  // Tổng hợp dữ liệu mỗi 5 giây
#define FLUSH_INTERVAL_MS       900000 // Ghi thẻ nhớ mỗi 15 phút (15 * 60 * 1000)

// Buffer lớn cho 15 phút dữ liệu
// 15 phút / 5 giây = 180 dòng. Mỗi dòng ~60 ký tự. 
// Tổng cần ~10.8KB. Cấp 12KB cho an toàn.
static char s_sd_buffer[12288] = {0}; 
static uint32_t s_buffer_len = 0;

// Moving Average Buffers
static float s_t_history[MA_WINDOW_SIZE] = {0};
static float s_h_history[MA_WINDOW_SIZE] = {0};
static int s_ma_head = 0;
static int s_ma_count = 0;

// ================= I2C INIT =================
static void init_i2c(void) {
    ESP_LOGI(TAG, "Initializing I2C Master...");
    i2c_config_t conf = {
        .mode = I2C_MODE_MASTER,
        .sda_io_num = I2C_MASTER_SDA_IO,
        .scl_io_num = I2C_MASTER_SCL_IO,
        .sda_pullup_en = GPIO_PULLUP_ENABLE,
        .scl_pullup_en = GPIO_PULLUP_ENABLE,
        .master.clk_speed = I2C_MASTER_FREQ_HZ,
    };
    i2c_param_config(I2C_MASTER_NUM, &conf);
    i2c_driver_install(I2C_MASTER_NUM, conf.mode, 0, 0, 0);
}

// ================= HELPER FUNCTIONS =================
static void lcd_show_status(void) {
    char buf[17];
    if (g_sensors.air_ok) {
        snprintf(buf, sizeof(buf), "T:%.1f H:%.1f%%", g_sensors.t_air, g_sensors.h_air);
    } else {
        snprintf(buf, sizeof(buf), "SENS ERROR ");
    }
    lcd_put_cur(0, 0);
    lcd_send_string(buf);

    const char *st_name = (g_current_state <= STATE_EMERGENCY) ? STATE_NAMES[g_current_state] : "UNK";
    snprintf(buf, sizeof(buf), "%-5s Set:%.0f%%", st_name, g_setpoints.target_hum);
    lcd_put_cur(1, 0);
    lcd_send_string(buf);
}

static void lcd_show_setup(void) {
    lcd_put_cur(0, 0); lcd_send_string("== SETUP MODE ==");
    char buf[17];
    snprintf(buf, sizeof(buf), "Target Hum: %.0f%%", g_setpoints.target_hum);
    lcd_put_cur(1, 0); lcd_send_string(buf);
}

// ================= MAIN APPLICATION =================
void app_main(void) {
    ESP_LOGI(TAG, "=== VXL SYSTEM STARTING (Smart Log) ===");

    // 1. Hardware Init
    init_i2c();
    lcd_init();
    sht35_init(I2C_MASTER_NUM, SHT35_ADDR_DEFAULT);
    ds18b20_init(PIN_DS18B20);
    mosfet_init(); 
    // Force Pull-up for buttons
    button_init(PIN_BTN_MODE);
    button_init(PIN_BTN_UP);
    button_init(PIN_BTN_DOWN);

    // SD Card Init & File Selection
    char log_filename[32] = {0};
    if (sd_card_init() == ESP_OK) {
        sd_get_next_filename("data", ".csv", log_filename, sizeof(log_filename));
        // CSV Header
        char header[] = "Filename,Time_ms,T_Air_Avg,H_Air_Avg,T_Water\n";
        sd_card_append_file(log_filename, header);
        ESP_LOGI(TAG, "Logging to: %s (Batch: 15min)", log_filename);
    } else {
        ESP_LOGW(TAG, "SD Card Missing!");
    }

    fsm_init();
    
    // Initial LCD
    lcd_clear();
    lcd_show_status();

    // Timing Variables
    uint32_t last_sensor_read = 0;
    uint32_t last_log_5s = 0;
    uint32_t last_flush_15m = xTaskGetTickCount() * portTICK_PERIOD_MS;
    
    // Button Logic
    int mode_hold_cnt = 0;
    bool mode_action_performed = false;

    while (1) {
        uint32_t now = xTaskGetTickCount() * portTICK_PERIOD_MS;

        // ============================================================ 
        // 1. FAST SENSOR SAMPLING & FILTERING (Every 250ms)
        // ============================================================ 
        if (!is_setup_mode && (now - last_sensor_read > SHT_SAMPLE_INTERVAL_MS)) {
            last_sensor_read = now;
            
            // A. Read SHT35
            sht35_start_measurement();
            // Blocking wait small enough for 250ms loop
            vTaskDelay(pdMS_TO_TICKS(20)); 
            
            float ta_raw, ha_raw;
            if (sht35_read_data(&ta_raw, &ha_raw) == ESP_OK) {
                // Add to Circular Buffer
                s_t_history[s_ma_head] = ta_raw;
                s_h_history[s_ma_head] = ha_raw;
                s_ma_head = (s_ma_head + 1) % MA_WINDOW_SIZE;
                if (s_ma_count < MA_WINDOW_SIZE) s_ma_count++;

                // Calculate Moving Average
                float t_sum = 0, h_sum = 0;
                for (int i=0; i<s_ma_count; i++) {
                    t_sum += s_t_history[i];
                    h_sum += s_h_history[i];
                }
                g_sensors.t_air = t_sum / s_ma_count;
                g_sensors.h_air = h_sum / s_ma_count;
                g_sensors.air_ok = true;
                g_sensors.error_count = 0;
            } else {
                g_sensors.error_count++;
                // Keep old values if failed briefly
                if (g_sensors.error_count > 10) g_sensors.air_ok = false;
            }

            // B. Trigger DS18B20 (Slow update, we read it later/async or just periodically)
            // Since we are in a fast loop, let's just trigger and read in the Logging block 
            // to avoid blocking 750ms here. Or read every 1s.
            // Simplified: We assume DS18B20 is handled less frequently or just read last known.
            // For now, let's read DS18B20 every 4th cycle (1 second) to be safe
            static int ds_div = 0;
            if (++ds_div >= 4) {
                float tw;
                if (ds18b20_read_temp(&tw) == ESP_OK) {
                    g_sensors.t_water = tw;
                    g_sensors.water_ok = true;
                }
                ds18b20_trigger_conversion();
                ds_div = 0;
            }
            
            // C. Run FSM Logic with new Averaged Data
            fsm_run_cycle(&g_sensors, &g_setpoints, &g_current_state, false);
            
            // D. Update LCD (Visual Feedback)
            lcd_show_status();
        }

        // ============================================================ 
        // 2. DATA LOGGING TO RAM BUFFER (Every 5s)
        // ============================================================ 
        if (!is_setup_mode && sd_card_is_mounted() && (now - last_log_5s > LOG_INTERVAL_MS)) {
            last_log_5s = now;

            // Prepare Data Line
            char line[128];
            float ta = g_sensors.air_ok ? g_sensors.t_air : -1.0;
            float ha = g_sensors.air_ok ? g_sensors.h_air : -1.0;
            float tw = g_sensors.water_ok ? g_sensors.t_water : -1.0;

            // Format: Filename, Time, T_Avg, H_Avg, T_Water
            int len = snprintf(line, sizeof(line), "%s,%lu,%.2f,%.2f,%.2f\n", 
                               log_filename, now, ta, ha, tw);
            
            // Append to RAM Buffer if space exists
            if (s_buffer_len + len < sizeof(s_sd_buffer) - 1) {
                memcpy(s_sd_buffer + s_buffer_len, line, len);
                s_buffer_len += len;
                s_sd_buffer[s_buffer_len] = '\0'; // Null terminate
                // ESP_LOGI(TAG, "RAM Buf: %d bytes", s_buffer_len);
            } else {
                ESP_LOGW(TAG, "RAM Buffer Full! Dropping sample.");
            }
        }

                // ============================================================
                // 3. FLUSH TO SD CARD (Every 15 Minutes)
                // ============================================================
                if (!is_setup_mode && sd_card_is_mounted() && (now - last_flush_15m > FLUSH_INTERVAL_MS)) {
                    last_flush_15m = now;
                    
                    if (s_buffer_len > 0) {
                        ESP_LOGW(TAG, ">>> FLUSHING %" PRIu32 " BYTES TO SD CARD...", s_buffer_len);
                        esp_err_t err = sd_card_append_file(log_filename, s_sd_buffer);
                        if (err == ESP_OK) {
                            ESP_LOGI(TAG, ">>> FLUSH SUCCESS. RAM CLEARED.");
                            s_buffer_len = 0;
                            s_sd_buffer[0] = '\0';
                        } else {
                            ESP_LOGE(TAG, ">>> FLUSH FAILED! Data retained.");
                            // Optional: Try to reconnect SD card?
                        }
                    }
                }
        // ============================================================ 
        // 4. BUTTON & MODE HANDLING
        // ============================================================ 
        if (button_is_down(PIN_BTN_MODE)) {
            mode_hold_cnt++;
            if (mode_hold_cnt >= 30 && !mode_action_performed) { // 3s hold
                mode_action_performed = true;
                
                if (!is_setup_mode) {
                    pwm_stop_all(); 
                    is_setup_mode = true;
                    lcd_clear(); lcd_show_setup();
                } else {
                    lcd_clear();
                    lcd_put_cur(0, 0); lcd_send_string("SAVING 5s...");
                    vTaskDelay(pdMS_TO_TICKS(5000));
                    
                    // Flush Log Buffer before reset (Optional but good)
                    if (s_buffer_len > 0 && sd_card_is_mounted()) {
                        sd_card_append_file(log_filename, s_sd_buffer);
                        s_buffer_len = 0;
                    }

                    // Reset MA Buffer logic
                    s_ma_count = 0; s_ma_head = 0;
                    
                    lcd_init(); 
                    fsm_init();
                    is_setup_mode = false;
                    lcd_clear();
                }
            }
        } else {
            mode_hold_cnt = 0;
            mode_action_performed = false;
        }

        if (is_setup_mode) {
            bool update = false;
            if (button_is_pressed(PIN_BTN_UP)) {
                g_setpoints.target_hum += 1.0f;
                if (g_setpoints.target_hum > 95.0f) g_setpoints.target_hum = 95.0f;
                update = true;
            }
            if (button_is_pressed(PIN_BTN_DOWN)) {
                g_setpoints.target_hum -= 1.0f;
                if (g_setpoints.target_hum < 5.0f) g_setpoints.target_hum = 5.0f;
                update = true;
            }
            if (update) lcd_show_setup();
        }

        // Main Loop Delay (Control Loop Speed)
        vTaskDelay(pdMS_TO_TICKS(100));
    }
}
