#include <stdio.h>
#include <string.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "esp_log.h"
#include "driver/i2c.h"
#include "driver/gpio.h"

// Include drivers
#include "config.h"
#include "lcd.h"
#include "sht35.h"
#include "pwm.h"
#include "sd.h"
// #include "ds18b20.h" // Removed for Blind Mode

#define TAG "TEST_MODE"

// hàm khởi tạo nhanh I2C
static void init_i2c_master(void) {
    i2c_config_t conf = {
        .mode = I2C_MODE_MASTER,
        .sda_io_num = I2C_MASTER_SDA_IO,
        .scl_io_num = I2C_MASTER_SCL_IO,
        .sda_pullup_en = GPIO_PULLUP_ENABLE,
        .scl_pullup_en = GPIO_PULLUP_ENABLE,
        .master.clk_speed = I2C_MASTER_FREQ_HZ,
    };
    i2c_param_config(I2C_MASTER_NUM, &conf);
    i2c_driver_install(I2C_MASTER_NUM, conf.mode, 0, 0, 0);
}

void app_main(void) {
    ESP_LOGI(TAG, "=== STARTING DE-HUMIDIFY (COOLING) TEST ===");

    // 1. INITIALIZATION
    init_i2c_master();
    lcd_init();
    sht35_init(I2C_MASTER_NUM, SHT35_ADDR_DEFAULT);
    // ds18b20_init(4); // Removed for Blind Mode
    mosfet_init();
    sd_card_init();

    // 2. PREPARE LOG FILE
    char filename[32] = {0};
    if (sd_card_is_mounted()) {
        sd_get_next_filename("test_dehum", ".csv", filename, sizeof(filename));
        ESP_LOGI(TAG, "Logging to: %s", filename);
        // Ghi Header
        sd_card_append_file(filename, "Time_ms,T_Air,H_Air,Pump_State,Peltier_Duty,CaseFan_Duty\n");
    }

    lcd_clear();
    lcd_send_string("MODE: DE-HUMID");
    lcd_put_cur(1, 0);
    lcd_send_string("Init Hardware...");
    vTaskDelay(pdMS_TO_TICKS(2000));

    // 3. CONFIGURATION - FIXED TIMER MODE (NO WATER SENSOR)
    const uint32_t RUN_DURATION_MS = 5 * 60 * 1000;   // Chạy 5 phút
    const uint32_t REST_DURATION_MS = 60 * 1000;      // Nghỉ 1 phút
    const uint32_t PRE_COOL_TIME_MS = 30 * 1000;      // 30s làm lạnh trước
    
    bool is_cooling_phase = true;
    uint32_t start_tick = xTaskGetTickCount() * portTICK_PERIOD_MS;

    // Initial Start
    ESP_LOGW(TAG, ">>> STARTING: FIXED TIMER MODE (5m RUN / 2m REST) <<<");
    mosfet_set_pump(false);         // Tắt bơm sục khí (vì đây là mode giảm ẩm)
    pwm_set_peltier(716, 1000);     // 70% Power (Mức tối ưu cho tản khí)
    pwm_set_case(1023, 1000);       // Fan nóng Max
    pwm_set_blower(0, 1000);        // OFF Blower (Chờ Pre-cool)
    pwm_set_circ(1023, 1000);       // Fan đối lưu

    // 4. MAIN LOOP
    char lcd_buf[32]; // Increased buffer size to avoid truncation warnings
    char log_buf[128];
    float t_air = 0, h_air = 0;

    while (1) {
        uint32_t now = xTaskGetTickCount() * portTICK_PERIOD_MS;
        uint32_t elapsed = now - start_tick;

        // --- CYCLE LOGIC ---
        if (is_cooling_phase) {
            // GIAI ĐOẠN CHẠY (RUN)
            if (elapsed > RUN_DURATION_MS) {
                is_cooling_phase = false;
                start_tick = now;
                ESP_LOGW(TAG, ">>> TIMER: End Run. Switch to Rest.");
                
                pwm_set_peltier(0, 1000);    // Tắt Peltier
                pwm_set_case(1023, 1000);    // Giữ quạt nóng xả nhiệt
                pwm_set_blower(0, 1000);     // Tắt quạt lạnh tránh bay hơi ngược
                
                lcd_clear();
                lcd_send_string("MODE: RESTING");
            } 
            else {
                if (elapsed < PRE_COOL_TIME_MS) {
                    // Pre-cool 60s
                    pwm_set_peltier(1023, 1000); // Ép lạnh nhanh lúc đầu
                    pwm_set_blower(0, 1000); 
                    
                    lcd_put_cur(0, 0);
                    snprintf(lcd_buf, sizeof(lcd_buf), "PRE-COOL | %ds ", (int)((PRE_COOL_TIME_MS - elapsed)/1000));
                    lcd_send_string(lcd_buf);
                } else {
                    // Running 4m còn lại
                    pwm_set_peltier(716, 1000);
                    pwm_set_blower(600, 1000);   // Gió chậm (60%) để ngưng tụ tốt
                    
                    lcd_put_cur(0, 0);
                    snprintf(lcd_buf, sizeof(lcd_buf), "RUNNING  | %dm ", (int)((RUN_DURATION_MS - elapsed)/60000));
                    lcd_send_string(lcd_buf);
                }
            }
        } else {
            // GIAI ĐOẠN NGHỈ (REST)
            if (elapsed > REST_DURATION_MS) {
                is_cooling_phase = true;
                start_tick = now;
                ESP_LOGW(TAG, ">>> TIMER: End Rest. Restarting Run.");
                
                pwm_set_peltier(1023, 1000); // Về lại Pre-cool
                pwm_set_blower(0, 1000); 
            } else {
                lcd_put_cur(0, 0);
                snprintf(lcd_buf, sizeof(lcd_buf), "COOLDOWN | %ds ", (int)((REST_DURATION_MS - elapsed)/1000));
                lcd_send_string(lcd_buf);
            }
        }

        // --- SENSORS & LOGGING ---
        sht35_start_measurement();
        vTaskDelay(pdMS_TO_TICKS(500)); 
        esp_err_t ret = sht35_read_data(&t_air, &h_air);
        
        if (ret == ESP_OK) {
            lcd_put_cur(1, 0);
            snprintf(lcd_buf, sizeof(lcd_buf), "T:%.1f H:%.1f    ", t_air, h_air);
            lcd_send_string(lcd_buf);

            if (sd_card_is_mounted() && filename[0] != 0) {
                snprintf(log_buf, sizeof(log_buf), "%lu,%.2f,%.2f,0,%d,1023\n", 
                         now, t_air, h_air, is_cooling_phase ? 716 : 0);
                sd_card_append_file(filename, log_buf);
            }
        }

        vTaskDelay(pdMS_TO_TICKS(2000));
    }
}
